cmake_minimum_required(VERSION 3.13) 
project(CFastMath)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED)

add_library(CFastMath STATIC 
      src/fastmath.c
      src/fastmath.h
)

target_compile_options(CFastMath PRIVATE 
      -O3 -ffast-math
      -fno-trapping-math 
      -flto -funroll-loops 
      -falign-functions=64 
      -falign-loops=64 -march=native
)
